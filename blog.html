Future<void> syncLocalToCloud() async {
  final pending = await _db.query(
    'tasks',
    where: 'sync_state = ?',
    whereArgs: ['DIRTY']
  );
  for (final row in pending) {
    try {
      await _cloudApi.upsert(row);
      await _db.update(
        'tasks',
        {'sync_state': 'SYNCED'},
        where: 'id = ?',
        whereArgs: [row['id']]
      );
    } on NetworkException {
      // Retain atomic local state
      break;
    }
  }
}
